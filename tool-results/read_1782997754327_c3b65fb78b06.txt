   434→function HomePage() {
   435→  const { setPage, addToCart, addToast, user, setUser } = useAppStore()
   436→  const [products, setProducts] = useState<HomeProduct[]>([])
   437→  const [promoProducts, setPromoProducts] = useState<HomeProduct[]>([])
   438→  const [activeTab, setActiveTab] = useState<'terbaru' | 'terlaris' | 'promo'>('terbaru')
   439→  const [loading, setLoading] = useState(true)
   440→  const [showBarcode, setShowBarcode] = useState(false)
   441→  const barcodeRef = useRef<SVGSVGElement>(null)
   442→
   443→  const memberCode = user ? `AGSI-${user.id.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 10)}` : ''
   444→
   445→  // Generate barcode when back side is visible
   446→  useEffect(() => {
   447→    if (showBarcode && barcodeRef.current && memberCode) {
   448→      try {
   449→        JsBarcode(barcodeRef.current, memberCode, {
   450→          format: 'CODE128',
   451→          width: 1.5,
   452→          height: 44,
   453→          displayValue: true,
   454→          font: 'monospace',
   455→          fontSize: 11,
   456→          textMargin: 4,
   457→          margin: 0,
   458→          background: 'transparent',
   459→          lineColor: '#ea580c',
   460→        })
   461→      } catch {}
   462→    }
   463→  }, [showBarcode, memberCode])
   464→
   465→  // Fetch latest user points/voucher on mount
   466→  useEffect(() => {
   467→    if (user && user.id) {
   468→      fetch(`/api/auth/profile?userId=${user.id}`)
   469→        .then((r) => r.json())
   470→        .then((data) => {
   471→          if (data.id) {
   472→            setUser({ ...user, points: data.points, voucher: data.voucher })
   473→          }
   474→        })
   475→        .catch(() => {})
   476→    }
   477→  }, [user?.id])
   478→
   479→  useEffect(() => {
   480→    Promise.all([
   481→      fetch('/api/products').then((r) => r.json()),
   482→      fetch('/api/products?tag=promo').then((r) => r.json()),
   483→    ])
   484→      .then(([all, promo]) => {
   485→        setProducts(Array.isArray(all) ? all : [])
   486→        setPromoProducts(Array.isArray(promo) ? promo : [])
   487→        setLoading(false)
   488→      })
   489→      .catch(() => {
   490→        setLoading(false)
   491→        addToast('Gagal memuat produk', 'error')
   492→      })
   493→  }, [addToast])
   494→
   495→  const handleAdd = (p: HomeProduct) => {
   496→    if (!user) {
   497→      addToast('Silakan login terlebih dahulu untuk menambahkan produk', 'error')
   498→      setPage('login')
   499→      return
   500→    }
   501→    addToCart({ productId: p.id, productName: p.name, price: p.price, quantity: 1, image: p.image })
   502→    addToast(`${p.name} ditambahkan ke keranjang`)
