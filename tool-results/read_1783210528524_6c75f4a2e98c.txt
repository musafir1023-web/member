   437→function HomePage() {
   438→  const { setPage, addToCart, addToast, user, setUser } = useAppStore()
   439→  const [products, setProducts] = useState<HomeProduct[]>([])
   440→  const [promoProducts, setPromoProducts] = useState<HomeProduct[]>([])
   441→  const [activeTab, setActiveTab] = useState<'terbaru' | 'terlaris' | 'promo'>('terbaru')
   442→  const [loading, setLoading] = useState(true)
   443→  const [showcaseFlipped, setShowcaseFlipped] = useState(false)
   444→  const showcaseBarcodeRef = useRef<SVGSVGElement>(null)
   445→
   446→  // Generate showcase barcode when back is visible
   447→  useEffect(() => {
   448→    if (showcaseFlipped && showcaseBarcodeRef.current) {
   449→      try {
   450→        JsBarcode(showcaseBarcodeRef.current, 'AGSI-RENCONG2025', {
   451→          format: 'CODE128', width: 1.5, height: 42, displayValue: true, font: 'monospace', fontSize: 10, textMargin: 3, margin: 0, background: 'transparent', lineColor: '#9A3412',
   452→        })
   453→      } catch {}
   454→    }
   455→  }, [showcaseFlipped])
   456→
   457→  // Fetch latest user points/voucher on mount
   458→  useEffect(() => {
   459→    if (user && user.id) {
   460→      fetch(`/api/auth/profile?userId=${user.id}`)
   461→        .then((r) => r.json())
   462→        .then((data) => {
   463→          if (data.id) {
   464→            setUser({ ...user, points: data.points, voucher: data.voucher })
   465→          }
   466→        })
   467→        .catch(() => {})
   468→    }
   469→  }, [user?.id])
   470→
   471→  useEffect(() => {
   472→    Promise.all([
   473→      fetch('/api/products').then((r) => r.json()),
   474→      fetch('/api/products?tag=promo').then((r) => r.json()),
   475→    ])
   476→      .then(([all, promo]) => {
   477→        setProducts(Array.isArray(all) ? all : [])
   478→        setPromoProducts(Array.isArray(promo) ? promo : [])
   479→        setLoading(false)
   480→      })
   481→      .catch(() => {
   482→        setLoading(false)
   483→        addToast('Gagal memuat produk', 'error')
   484→      })
   485→  }, [addToast])
   486→
   487→  const handleAdd = (p: HomeProduct) => {
   488→    if (!user) {
   489→      addToast('Silakan login terlebih dahulu untuk menambahkan produk', 'error')
   490→      setPage('login')
   491→      return
   492→    }
   493→    addToCart({ productId: p.id, productName: p.name, price: p.price, quantity: 1, image: p.image })
   494→    addToast(`${p.name} ditambahkan ke keranjang`)
   495→  }
   496→
