   431→function HomePage() {
   432→  const { setPage, addToCart, addToast, user, setUser } = useAppStore()
   433→  const [products, setProducts] = useState<HomeProduct[]>([])
   434→  const [promoProducts, setPromoProducts] = useState<HomeProduct[]>([])
   435→  const [activeTab, setActiveTab] = useState<'terbaru' | 'terlaris' | 'promo'>('terbaru')
   436→  const [loading, setLoading] = useState(true)
   437→  const [showBarcode, setShowBarcode] = useState(false)
   438→  const barcodeRef = useRef<SVGSVGElement>(null)
   439→
   440→  const memberCode = user ? `AGSI-${user.id.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 10)}` : ''
   441→
   442→  // Generate barcode when back side is visible
   443→  useEffect(() => {
   444→    if (showBarcode && barcodeRef.current && memberCode) {
   445→      try {
   446→        JsBarcode(barcodeRef.current, memberCode, {
   447→          format: 'CODE128',
   448→          width: 1.5,
   449→          height: 44,
   450→          displayValue: true,
   451→          font: 'monospace',
   452→          fontSize: 11,
   453→          textMargin: 4,
   454→          margin: 0,
   455→          background: 'transparent',
   456→          lineColor: '#ea580c',
   457→        })
   458→      } catch {}
   459→    }
   460→  }, [showBarcode, memberCode])
   461→
   462→  // Fetch latest user points/voucher on mount
   463→  useEffect(() => {
   464→    if (user && user.id) {
   465→      fetch(`/api/auth/profile?userId=${user.id}`)
   466→        .then((r) => r.json())
   467→        .then((data) => {
   468→          if (data.id) {
   469→            setUser({ ...user, points: data.points, voucher: data.voucher })
   470→          }
   471→        })
   472→        .catch(() => {})
   473→    }
   474→  }, [user?.id])
   475→
   476→  useEffect(() => {
   477→    Promise.all([
   478→      fetch('/api/products').then((r) => r.json()),
   479→      fetch('/api/products?tag=promo').then((r) => r.json()),
   480→    ])
   481→      .then(([all, promo]) => {
   482→        setProducts(all)
   483→        setPromoProducts(promo)
   484→        setLoading(false)
   485→      })
   486→      .catch(() => {
   487→        setLoading(false)
   488→        addToast('Gagal memuat produk', 'error')
   489→      })
   490→  }, [addToast])
   491→
   492→  const handleAdd = (p: HomeProduct) => {
   493→    if (!user) {
   494→      addToast('Silakan login terlebih dahulu untuk menambahkan produk', 'error')
   495→      setPage('login')
   496→      return
   497→    }
   498→    addToCart({ productId: p.id, productName: p.name, price: p.price, quantity: 1, image: p.image })
   499→    addToast(`${p.name} ditambahkan ke keranjang`)
   500→  }
   501→
