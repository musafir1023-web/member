   438→/* ─────────────────────── MEMBER FLIP CARD ─────────────────────── */
   439→function MemberFlipCard({ user }: { user: { id: string; name: string; email: string } }) {
   440→  const [flipped, setFlipped] = useState(false)
   441→
   442→  const memberId = `AGSI-${user.id.slice(0, 4).toUpperCase()}-${user.id.slice(4, 8).toUpperCase()}`
   443→  const points = 1250
   444→  const vouchers = 3
   445→
   446→  // Generate deterministic barcode bars from user id
   447→  const bars = user.id.split('').map((c) => c.charCodeAt(0)).flatMap((n, i) => {
   448→    const seed = (n * (i + 7)) % 17
