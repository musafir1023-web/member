   634→/* ─────────────────────── CHECKOUT FORM ─────────────────────── */
   635→function CheckoutForm({ onCancel }: { onCancel: () => void }) {
   636→  const { cart, getCartTotal, clearCart, setPage, addToast, user, setReceipt } = useAppStore()
   637→  const total = useAppStore(getCartTotal)
   638→  const [submitting, setSubmitting] = useState(false)
   639→  const [form, setForm] = useState({
   640→    customerName: user?.name || '',
   641→    customerPhone: user?.phone || '',
   642→    customerAddress: '',
   643→    notes: '',
   644→    paymentMethod: 'COD',
   645→  })
   646→
   647→  const updateField = (field: string, value: string) => setForm((prev) => ({ ...prev, [field]: value }))
   648→
   649→  const handleSubmit = async (e: React.FormEvent) => {
   650→    e.preventDefault()
   651→    if (!form.customerName || !form.customerPhone || !form.customerAddress) {
   652→      addToast('Mohon lengkapi semua data yang diperlukan', 'error')
   653→      return
   654→    }
   655→    setSubmitting(true)
   656→
   657→    try {
   658→      const items = cart.map((c) => ({
   659→        productId: c.productId,
   660→        productName: c.productName,
   661→        quantity: c.quantity,
   662→        price: c.price,
   663→        subtotal: c.price * c.quantity,
   664→      }))
   665→
   666→      const res = await fetch('/api/orders', {
   667→        method: 'POST',
   668→        headers: { 'Content-Type': 'application/json' },
   669→        body: JSON.stringify({
   670→          userId: user?.id || null,
   671→          items,
   672→          customerName: form.customerName,
   673→          customerPhone: form.customerPhone,
   674→          customerAddress: form.customerAddress,
   675→          notes: form.notes,
   676→          paymentMethod: form.paymentMethod,
   677→        }),
   678→      })
   679→
   680→      const data = await res.json()
   681→      if (!res.ok) throw new Error(data.error || 'Gagal membuat pesanan')
   682→
   683→      setReceipt(data)
   684→      clearCart()
   685→      addToast('Pesanan berhasil dibuat!', 'success')
   686→      setPage('receipt')
   687→    } catch {
   688→      addToast('Gagal membuat pesanan. Silakan coba lagi.', 'error')

   690→      setSubmitting(false)
   691→    }
