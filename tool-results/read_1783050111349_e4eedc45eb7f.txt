   437→function HomePage() {
   438→  const { setPage, addToCart, addToast, user, setUser } = useAppStore()
   439→  const [products, setProducts] = useState<HomeProduct[]>([])
   440→  const [promoProducts, setPromoProducts] = useState<HomeProduct[]>([])
   441→  const [activeTab, setActiveTab] = useState<'terbaru' | 'terlaris' | 'promo'>('terbaru')
   442→  const [loading, setLoading] = useState(true)
   443→  const [showBarcode, setShowBarcode] = useState(false)
   444→  const barcodeRef = useRef<SVGSVGElement>(null)
   445→
   446→  const memberCode = user ? `AGSI-${user.id.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 10)}` : ''
   447→
   448→  // Generate barcode when back side is visible
   449→  useEffect(() => {
   450→    if (showBarcode && barcodeRef.current && memberCode) {
   451→      try {
   452→        JsBarcode(barcodeRef.current, memberCode, {
   453→          format: 'CODE128',
   454→          width: 1.5,
   455→          height: 44,
   456→          displayValue: true,
   457→          font: 'monospace',
   458→          fontSize: 11,
   459→          textMargin: 4,
   460→          margin: 0,
   461→          background: 'transparent',
   462→          lineColor: '#ea580c',
   463→        })
   464→      } catch {}
   465→    }
   466→  }, [showBarcode, memberCode])
   467→
   468→  // Fetch latest user points/voucher on mount
   469→  useEffect(() => {
   470→    if (user && user.id) {
   471→      fetch(`/api/auth/profile?userId=${user.id}`)
   472→        .then((r) => r.json())
   473→        .then((data) => {
   474→          if (data.id) {
   475→            setUser({ ...user, points: data.points, voucher: data.voucher })
   476→          }
   477→        })
   478→        .catch(() => {})
   479→    }
   480→  }, [user?.id])
   481→
   482→  useEffect(() => {
   483→    Promise.all([
   484→      fetch('/api/products').then((r) => r.json()),
   485→      fetch('/api/products?tag=promo').then((r) => r.json()),
   486→    ])
   487→      .then(([all, promo]) => {
   488→        setProducts(Array.isArray(all) ? all : [])
   489→        setPromoProducts(Array.isArray(promo) ? promo : [])
   490→        setLoading(false)
   491→      })
   492→      .catch(() => {
   493→        setLoading(false)
   494→        addToast('Gagal memuat produk', 'error')
   495→      })
   496→  }, [addToast])
   497→
   498→  const handleAdd = (p: HomeProduct) => {
   499→    if (!user) {
   500→      addToast('Silakan login terlebih dahulu untuk menambahkan produk', 'error')
   501→      setPage('login')
   502→      return
