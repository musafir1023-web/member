   452→  const [promoProducts, setPromoProducts] = useState<HomeProduct[]>([])

   455→  const [showBarcode, setShowBarcode] = useState(false)
   456→  const barcodeRef = useRef<SVGSVGElement>(null)
   457→
   458→  const memberCode = user ? `AGSI-${user.id.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 10)}` : ''
   459→
   460→  // Generate barcode when back side is visible
   461→  useEffect(() => {
   462→    if (showBarcode && barcodeRef.current && memberCode) {
   463→      try {
   464→        JsBarcode(barcodeRef.current, memberCode, {
   465→          format: 'CODE128',
   466→          width: 1.5,
   467→          height: 44,
   468→          displayValue: true,
   469→          font: 'monospace',
   470→          fontSize: 11,
   471→          textMargin: 4,
   472→          margin: 0,
   473→          background: 'transparent',
   474→          lineColor: '#ea580c',
   475→        })
   476→      } catch {}
   477→    }
   478→  }, [showBarcode, memberCode])
   479→
   480→  // Fetch latest user points/voucher on mount
   481→  useEffect(() => {
   482→    if (user && user.id) {
   483→      fetch(`/api/auth/profile?userId=${user.id}`)
   484→        .then((r) => r.json())
   485→        .then((data) => {
   486→          if (data.id) {
   487→            setUser({ ...user, points: data.points, voucher: data.voucher })
   488→          }
   489→        })
   490→        .catch(() => {})
   491→    }
   492→  }, [user?.id])
   493→
