   351→interface Product {

   355→  price: number

   361→function MenuPage() {
   362→  const [products, setProducts] = useState<Product[]>([])
   363→  const [loading, setLoading] = useState(true)
   364→  const [activeCategory, setActiveCategory] = useState('Semua')
   365→  const addToCart = useAppStore((s) => s.addToCart)
   366→  const addToast = useAppStore((s) => s.addToast)
   367→
   368→  useEffect(() => {
   369→    fetch('/api/products')
   370→      .then((r) => r.json())
   371→      .then((data) => {
   372→        setProducts(data)
   373→        setLoading(false)
   374→      })
   375→      .catch(() => {
   376→        setLoading(false)
   377→        addToast('Gagal memuat menu', 'error')
   378→      })
   379→  }, [addToast])
   380→
   381→  const categories = ['Semua', ...Array.from(new Set(products.map((p) => p.category)))]
   382→  const filtered = activeCategory === 'Semua' ? products : products.filter((p) => p.category === activeCategory)
   383→
   384→  const handleAdd = (p: Product) => {
   385→    addToCart({ productId: p.id, productName: p.name, price: p.price, quantity: 1, image: p.image })

   388→
   389→  return (
   390→    <div className="min-h-screen">
   391→      {/* Hero mini */}

   405→          {categories.map((cat) => (
   406→            <Button
   407→              key={cat}
   408→              size="sm"
   409→              variant={activeCategory === cat ? 'default' : 'outline'}

   420→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   421→            {[...Array(6)].map((_, i) => (
   422→              <div key={i} className="bg-white rounded-xl overflow-hidden shadow-md">

   580→function CheckoutForm({ onCancel }: { onCancel: () => void }) {
   581→  const { cart, getCartTotal, clearCart, setPage, addToast, user, setReceipt } = useAppStore()
   582→  const total = useAppStore(getCartTotal)
   583→  const [submitting, setSubmitting] = useState(false)
   584→  const [form, setForm] = useState({
   585→    customerName: user?.name || '',
   586→    customerPhone: user?.phone || '',

   588→    notes: '',

   590→  })
   591→
   592→  const updateField = (field: string, value: string) => setForm((prev) => ({ ...prev, [field]: value }))
   593→
   594→  const handleSubmit = async (e: React.FormEvent) => {
   595→    e.preventDefault()
   596→    if (!form.customerName || !form.customerPhone || !form.customerAddress) {
   597→      addToast('Mohon lengkapi semua data yang diperlukan', 'error')
   598→      return
   599→    }

   601→
   602→    try {
   603→      const items = cart.map((c) => ({
   604→        productId: c.productId,
   605→        productName: c.productName,
   606→        quantity: c.quantity,
   607→        price: c.price,
   608→        subtotal: c.price * c.quantity,
   609→      }))
   610→
   611→      const res = await fetch('/api/orders', {
   612→        method: 'POST',
   613→        headers: { 'Content-Type': 'application/json' },

   615→          userId: user?.id || null,
   616→          items,
   617→          customerName: form.customerName,
   618→          customerPhone: form.customerPhone,
   619→          customerAddress: form.customerAddress,
   620→          notes: form.notes,
   621→          paymentMethod: form.paymentMethod,
   622→        }),
   623→      })
   624→
   625→      const data = await res.json()
   626→      if (!res.ok) throw new Error(data.error || 'Gagal membuat pesanan')
   627→
   628→      setReceipt(data)
   629→      clearCart()
   630→      addToast('Pesanan berhasil dibuat!', 'success')
   631→      setPage('receipt')
   632→    } catch {
   633→      addToast('Gagal membuat pesanan. Silakan coba lagi.', 'error')
